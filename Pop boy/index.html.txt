<script>
const img = document.getElementById("boy");
const score = document.getElementById("score");
const soundBtn = document.getElementById("soundBtn");

let clicks = Number(localStorage.getItem("popboy-score") || 0);
let soundOn = true;
score.textContent = clicks;

function pop(){
    clicks++;
    score.textContent = clicks;
    localStorage.setItem("popboy-score", clicks);

    if(soundOn){
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const o = ctx.createOscillator();
        const g = ctx.createGain();
        o.type = "square";
        o.frequency.value = 650;
        o.connect(g);
        g.connect(ctx.destination);
        g.gain.value = 0.08;
        o.start();
        o.stop(ctx.currentTime + 0.05);
    }
}

function openMouth(){
    img.src = "open.jpg";
}

function closeMouth(){
    img.src = "normal.jpg";
}

img.addEventListener("mousedown", openMouth);
img.addEventListener("mouseup", closeMouth);
img.addEventListener("mouseleave", closeMouth);

img.addEventListener("touchstart", (e)=>{
    e.preventDefault();
    openMouth();
    pop();
}, {passive:false});

img.addEventListener("touchend", closeMouth);

img.addEventListener("click", pop);

soundBtn.addEventListener("click", ()=>{
    soundOn = !soundOn;
    soundBtn.innerHTML = soundOn ? "🔊 Sound: ON" : "🔇 Sound: OFF";
});
</script>
